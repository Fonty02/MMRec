\begin{frame}{Recommender Systems}
  \begin{itemize}
    \item Guide users to relevant items in a large space \cite{Lops2011}
    \item Approaches: Content-Based, Collaborative, Hybrid, Knowledge-Based, Context-Aware
  \end{itemize}
  \begin{figure}[h]
    \centering
    \includegraphics[height=0.4\textheight]{../documentazione/img/RecSys.png}
    \caption{Example of a Recommender System}
  \end{figure}

\begin{frame}{Multimodal Recommender Systems}
  \begin{itemize}
    \item Use multiple modalities (text, image, audio) to enrich item representations \cite{Spillong}
    \item Encoding options: concat, fusion/attention, GNNs \cite{ConcatFeature, Fusion}
  \end{itemize}
  \begin{figure}[h]
    \centering
    \includegraphics[height=0.4\textheight]{img/StarWars.jpg}
    \caption{Example of multimodal data for a movie (Star Wars)}
  \end{figure}
\end{frame}


\begin{frame}{Motivation \& Problem Statement}
  
    Traditionally, MMRec systems extract features using independent encoders (e.g., BERT for text, ResNet for images). In other domains, like Few-Shot Learning for Image Classification, aligned features (CLIP) have shown superior performance with respect to disaligned ones.
\end{frame}

% ------------------------- Aligned Models -------------------------
\section{Aligned Models}
\begin{frame}{Contrastive Learning}
  \begin{columns}
      \begin{column}{0.6\textwidth}
          \begin{itemize}
            \item Self-supervised method: learns representations that bring similar samples closer \cite{ContrastiveLearning}
            \item \textbf{Goal:} Minimize distance between positive pairs, maximize distance between negative pairs.
            \item Contrastive loss: 
            $$ \ell_{i,j} = -\log \frac{\exp(\text{sim}(z_i, z_j)/\tau)}{\sum_{k=1}^{2N} [k \neq i] \exp(\text{sim}(z_i, z_k)/\tau)} $$
          \end{itemize}
      \end{column}
      \begin{column}{0.4\textwidth}
          \begin{figure}[h]
            \centering
            \includegraphics[width=\textwidth]{../documentazione/img/CLearning.png}
          \end{figure}
      \end{column}
  \end{columns}
\end{frame}

\begin{frame}{CLIP (Contrastive Language-Image Pre-training)}
  \begin{itemize}
    \item Aligns image and text embeddings in a shared space (OpenAI) \cite{CLIP}
    \item \textbf{Relevance:} Large multimodal encoders like CLIP can improve performance when fine-tuned end-to-end \cite{CLIPRecSys}
  \end{itemize}
  \begin{figure}[h]
    \centering
    \includegraphics[height=0.4\textheight]{../documentazione/img/CLIP.png}
    \caption{CLIP aligns visual and textual concepts}
  \end{figure}
\end{frame}

\begin{frame}{AudioCLIP}
  \begin{itemize}
    \item Extends CLIP by adding an \textbf{audio encoder} (ESResNeXt) \cite{AudioCLIP}
    \item \textbf{Training process:}
    \begin{enumerate}
        \item Audio-head pre-training
        \item Cooperative distillation (aligning audio to frozen CLIP space)
        \item Full tri-modal fine-tuning on AudioSet
    \end{enumerate}
  \end{itemize}
  \begin{figure}[h]
    \centering
    \includegraphics[height=0.3\textheight]{../documentazione/img/AudioCLIP.png}
    \caption{Tri-modal alignment in AudioCLIP}
  \end{figure}
\end{frame}

\section{Experiments}
\begin{frame}{Embedding Distribution Analysis (UMAP + KDE)}
  To understand \textit{why} models perform differently, we analyzed the geometric structure of the embeddings.
  \vspace{1em}
  \begin{itemize}
    \item \textbf{Technique:} UMAP projection to 2D $\rightarrow$ S1 Normalization (Unit Circle) $\rightarrow$ Gaussian KDE.
    \item \textbf{What to look for:}
    \begin{itemize}
        \item \textit{Uniformity:} Do features cover the space?
        \item \textit{Peaks:} Distinct peaks indicate specific feature orientations.
        \item \textit{Collapse:} Overlapping distributions imply loss of modality-specific signal.
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{MovieLens 1M: Aligned vs Disaligned}
    \begin{columns}
        \begin{column}{0.5\textwidth}

    \end{itemize}

% ------------------------- Discussion & Conclusion -------------------------
\section{Discussion \& Conclusion}

\begin{frame}{Discussion: Context Matters}
  \begin{block}{MovieLens 1M (Movies)}
    \begin{itemize}
        \item Modalities (Posters, Plot) share high semantic correlation.
        \item \textbf{Result:} Alignment (CLIP) helps. It reinforces the shared signal.
    \end{itemize}
  \end{block}
  
  \begin{block}{LastFM (Music)}
    \begin{itemize}
        \item Modalities (Audio, Cover Art) are semantically distinct/orthogonal.
        \item \textbf{Result:} Strict alignment hurts. Disaligned features (keeping specific audio/visual signals separate) perform better.
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Conclusions and Future Work}
  \begin{itemize}
    \item \textbf{Aggressive Alignment (AudioCLIP):} Can be detrimental if it collapses the feature space too much, erasing fine-grained details needed for ranking.
    \item \textbf{Simplicity Wins:} Simple projection models (VBPR) with high-quality features often beat complex GNNs (LATTICE) on these datasets.
    \item \textbf{Future Directions:}
    \begin{itemize}
        \item Develop domain-adaptive fusion strategies (soft-alignment).
        \item Investigate regularization techniques to prevent feature collapse in tri-modal alignment.
    \end{itemize}
  \end{itemize}
\end{frame}
